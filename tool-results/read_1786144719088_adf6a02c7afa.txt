   537→<script>
   538→  const form = document.getElementById("contact-form") as HTMLFormElement | null;
   539→  const submitBtn = document.getElementById("contact-submit") as HTMLButtonElement | null;
   540→  const submitLabel = document.getElementById("submit-label");
   541→  const submitLoading = document.getElementById("submit-loading");
   542→  const successEl = document.getElementById("form-success");
   543→  const errorEl = document.getElementById("form-error");
   544→  const errorText = document.getElementById("form-error-text");
   545→
   546→  function setLoading(loading: boolean) {
   547→    if (!submitBtn || !submitLabel || !submitLoading) return;
   548→    submitBtn.disabled = loading;
   549→    submitLabel.classList.toggle("hidden", loading);
   550→    submitLoading.classList.toggle("hidden", !loading);
   551→  }
   552→
   553→  form?.addEventListener("submit", async (e) => {
   554→    e.preventDefault();
   555→    successEl?.classList.add("hidden");
   556→    errorEl?.classList.add("hidden");
   557→    const formData = new FormData(form);
   558→    setLoading(true);
   559→    try {
   560→      const res = await fetch("/api/contact", {
   561→        method: "POST",
   562→        headers: { "Content-Type": "application/json" },
   563→        body: JSON.stringify({
   564→          name: formData.get("name"),
   565→          email: formData.get("email"),
   566→          phone: formData.get("phone"),
   567→          message: formData.get("message"),
   568→        }),
   569→      });
   570→      const json = await res.json().catch(() => ({}));
   571→      if (res.ok && json.success) {
   572→        successEl?.classList.remove("hidden");
   573→        errorEl?.classList.add("hidden");
   574→        form.reset();
   575→      } else {
   576→        errorEl?.classList.remove("hidden");
   577→        successEl?.classList.add("hidden");
   578→        if (errorText && json.error) errorText.textContent = json.error;
   579→      }
   580→    } catch {
   581→      errorEl?.classList.remove("hidden");
   582→      successEl?.classList.add("hidden");
   583→    } finally {
   584→      setLoading(false);
   585→    }
   586→  });
   587→
   588→  // Smooth scroll for anchor links
   589→  document.querySelectorAll('a[href^="#"]').forEach(anchor => {
   590→    anchor.addEventListener('click', function (e) {
   591→      const href = (this as HTMLAnchorElement).getAttribute('href');
   592→      if (href === '#') return;
   593→      e.preventDefault();
   594→      const target = document.querySelector(href);
   595→      if (target) {
   596→        target.scrollIntoView({ behavior: 'smooth' });
   597→        // Close mobile menu if open
   598→        const navToggle = document.getElementById('nav-toggle') as HTMLInputElement | null;
   599→        if (navToggle?.checked) navToggle.checked = false;
   600→      }
   601→    });
   602→  });

    30→function escHtml(s: string): string {
    31→  return s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
    32→}
    33→
    34→function allowHtml(s: string): string {
    35→  // Content from DB already has HTML like <strong>, so just escape & then allow safe tags
    36→  let out = s.replace(/&/g, "&amp;");
    37→  // Don't double-escape existing HTML entities or tags
    38→  // Actually the data comes with raw <strong> etc. So just return as-is but escape bare &
    39→  return out;
    40→}

   940→  const AUTH_PASS = "admin123456";
   941→
   942→  function authenticate(req: express.Request, res: express.Response, next: express.NextFunction) {
   943→    const token = req.headers.authorization?.replace("Bearer ", "");
   944→    if (token && sessions.has(token)) return next();
   945→    res.status(401).json({ error: "Unauthorized" });
   946→  }
   947→
   948→  app.post("/api/auth/login", (req, res) => {
   949→    const { email, password } = req.body || {};
   950→    if (email === AUTH_USER && password === AUTH_PASS) {
   951→      const token = Math.random().toString(36).slice(2);
   952→      sessions.set(token, true);
   953→      res.json({ token, user: { email } });
   954→    } else {
   955→      res.status(401).json({ error: "Invalid credentials" });
   956→    }
   957→  });

   960→    try {
   961→      const { collection } = req.params;
   962→      const limit = parseInt(req.query.limit as string) || 10;
   963→      const page = parseInt(req.query.page as string) || 1;
   964→      const result = await payload.find({ collection, limit, page });
   965→      res.json({
   966→        docs: result.docs, totalDocs: result.totalDocs, limit: result.limit,
   967→        totalPages: result.totalPages, page: result.page, pagingCounter: result.pagingCounter,
   968→        hasPrevPage: result.hasPrevPage, hasNextPage: result.hasNextPage,
   969→        prevPage: result.prevPage, nextPage: result.nextPage,
   970→      });
   971→    } catch (err: any) { res.status(500).json({ error: err.message }); }
   972→  });

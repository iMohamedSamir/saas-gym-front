    30→    30→function escHtml(s: string): string {
    31→    31→  return s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
    32→    32→}
    33→    33→
    34→    34→function allowHtml(s: string): string {
    35→    35→  // Content from DB already has HTML like <strong>, so just escape & then allow safe tags
    36→    36→  let out = s.replace(/&/g, "&amp;");
    37→    37→  // Don't double-escape existing HTML entities or tags
    38→    38→  // Actually the data comes with raw <strong> etc. So just return as-is but escape bare &
    39→    39→  return out;
    40→    40→}
